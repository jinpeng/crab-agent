```rust
fn parse_frontmatter(content: &str) -> Option<(Option<String>, Option<String>)> {
    let mut lines = content.lines();
    if lines.next()?.trim_end() != "---" {
        return None;
    }
    let mut name = None;
    let mut description = None;
    for line in lines {
        let line = line.trim_end();
        if line == "---" {
            return Some((name, description));
        }
        if let Some((k, v)) = line.split_once(':') {
            let v = v.trim();
            match k.trim() {
                "name" => name = Some(v.to_string()),
                "description" => description = Some(v.to_string()),
                _ => {} // license/metadata/allowed-tools 等一律忽略（YAGNI）
            }
        }
    }
    None // 没有闭合 ---：视为无 frontmatter
}

pub fn discover(cwd: &Path, home: &Path) -> Vec<Skill> {
    let mut skills = Vec::new();
    let mut seen_dirs: HashSet<PathBuf> = HashSet::new();
    let mut seen_names: HashSet<String> = HashSet::new();

    // 项目级：cwd 向上逐层，含 git repo 根本身后停止。
    let mut dir = cwd.to_path_buf();
    loop {
        scan_dir(
            &dir.join(".agents/skills"),
            &mut skills,
            &mut seen_dirs,
            &mut seen_names,
        );
        if dir.join(".git").exists() {
            break;
        }
        match dir.parent() {
            Some(p) if p != dir => dir = p.to_path_buf(),
            _ => break,
        }
    }
    // 用户级最后：项目级（近 > 远）优先于用户级。
    scan_dir(
        &home.join(".agents/skills"),
        &mut skills,
        &mut seen_dirs,
        &mut seen_names,
    );

    skills.sort_by(|a, b| a.name.cmp(&b.name));
    skills
}

fn scan_dir(
    skills_dir: &Path,
    skills: &mut Vec<Skill>,
    seen_dirs: &mut HashSet<PathBuf>,
    seen_names: &mut HashSet<String>,
) {
    let Ok(entries) = std::fs::read_dir(skills_dir) else {
        return; // 目录不存在：正常，不是错误
    };
    // 排序保证发现顺序（同名冲突谁赢）确定。
    let mut dirs: Vec<PathBuf> = entries.flatten().map(|e| e.path()).filter(|p| p.is_dir()).collect();
    dirs.sort();
    for dir in dirs {
        let skill_md = dir.join("SKILL.md");
        if !skill_md.is_file() {
            continue;
        }
        let canon = std::fs::canonicalize(&dir).unwrap_or_else(|_| dir.clone());
        if !seen_dirs.insert(canon) {
            continue;
        }
        let label = dir.display();
        let Ok(content) = std::fs::read_to_string(&skill_md) else {
            eprintln!("[crab] skill '{label}': SKILL.md not valid UTF-8 (skipped)");
            continue;
        };
        let Some((fm_name, fm_desc)) = parse_frontmatter(&content) else {
            eprintln!("[crab] skill '{label}': no frontmatter (skipped)");
            continue;
        };
        let Some(description) = fm_desc.filter(|d| !d.is_empty()) else {
            eprintln!("[crab] skill '{label}': no description (skipped)");
            continue;
        };
        let name = fm_name.filter(|n| !n.is_empty()).unwrap_or_else(|| {
            dir.file_name()
                .map(|n| n.to_string_lossy().into_owned())
                .unwrap_or_default()
        });
        if !seen_names.insert(name.clone()) {
            eprintln!("[crab] skill '{name}' in {label}: duplicate name (skipped, first found wins)");
            continue;
        }
        skills.push(Skill { name, description, dir, skill_md });
    }
}
```

```rust
pub fn build_system_prompt(
    cwd: &Path,
    extra: Option<String>,
    skills: &[crab_core::skills::Skill],
) -> String {
    let mut sp = String::from(BASE_PROMPT);
    if let Ok(content) = std::fs::read_to_string(cwd.join("AGENTS.md")) {
        sp.push_str("\n\n# Project instructions (AGENTS.md)\n");
        sp.push_str(&content);
    }
    if !skills.is_empty() {
        sp.push_str(
            "\n\n# Skills\nAvailable skills. When a task matches a skill's description, \
             read its SKILL.md at the given path with the read tool and follow it. \
             Bundled files (references/, scripts/, assets/) are relative to the skill's directory.\n",
        );
        for s in skills {
            sp.push_str(&format!("- {} — {} (at {})\n", s.name, s.description, s.skill_md.display()));
        }
        sp.pop(); // 去掉末行换行，保持段落整洁
    }
    if let Some(e) = extra {
        sp.push_str("\n\n# Additional instructions\n");
        sp.push_str(&e);
    }
    sp
}
```

```rust
/// Build the user message that injects a skill's instructions. `args` (may be
/// empty) is appended as the user request after the skill block.
pub fn skill_message(skill: &crab_core::skills::Skill, args: &str) -> String {
    let content = std::fs::read_to_string(&skill.skill_md).unwrap_or_else(|e| {
        format!("(failed to read {}: {e})", skill.skill_md.display())
    });
    let mut msg = format!(
        "<Skill instructions: {} ({})>\n{}\n</Skill instructions>",
        skill.name,
        skill.skill_md.display(),
        content
    );
    if !args.is_empty() {
        msg.push_str("\n\n");
        msg.push_str(args);
    }
    msg
}
```
